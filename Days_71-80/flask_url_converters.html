intIntegerWhole numbers only, no decimals
@app.route('/post/<int:id>')
def get_post(id):          # id is already an int
    return f'Post #{id}'

# /post/42   → id = 42  ✓
# /post/abc  → 404 automatically ✗
floatFloatDecimal numbers
@app.route('/price/<float:amount>')
def show_price(amount):
    return f'Price: ${amount:.2f}'

# /price/9.99  → amount = 9.99  ✓
# /price/abc   → 404 automatically ✗
stringString (default)Anything except a slash
@app.route('/user/<string:name>')
def profile(name):
    return f'User: {name}'

# <name> and <string:name> are identical
# /user/alice → name = "alice"  ✓
# /user/a/b   → 404 (slash not allowed) ✗
pathPathLike string, but slashes allowed
@app.route('/files/<path:filepath>')
def get_file(filepath):
    return f'File: {filepath}'

# /files/docs/report.pdf → filepath = "docs/report.pdf"  ✓
uuidUUIDValidates UUID format strictly
@app.route('/item/<uuid:item_id>')
def get_item(item_id):   # item_id is a uuid.UUID object
    return f'Item: {item_id}'

# /item/123e4567-e89b-12d3-a456-426614174000  ✓
# /item/not-a-uuid                            → 404 ✗
Why this matters
Without a converter, /post/abc would reach your function as the string "abc" and crash when you try to use it as an ID. With <int:id>, Flask rejects invalid values with a 404 before your code even runs.
