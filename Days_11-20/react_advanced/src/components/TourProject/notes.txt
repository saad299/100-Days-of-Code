      import { useEffect, useState } from "react";

const API_KEY = "ZtyH8c-n1NZfDQ9K3MjFehcX1nTIzXRZPOEXbj4gFYI";
export default function Hero() {
  const [images, setImages] = useState([]);

  // Fetch Unsplash images
  useEffect(() => {
    fetch(
      `https://api.unsplash.com/photos/random?count=6&query=nature&client_id=${API_KEY}`
    )
      .then((res) => res.json())
      .then((data) => {
        setImages(data.map((img) => img.urls.regular));
      });
  }, []);

  // rotate right
  const handleNext = () => {
    setImages((prev) => [...prev.slice(1), prev[0]]);
  };

  // rotate left
  const handlePrev = () => {
    setImages((prev) => [prev[prev.length - 1], ...prev.slice(0, -1)]);
  };

  return (
    <div className="mt-20 w-full h-[700px] bg-gray-200 shadow-2xl mx-auto overflow-hidden">
      <div id="carousel" className="relative h-full">
        {images.map((src, i) => (
          <div
            key={i}
            className="item absolute top-1/2 -translate-y-1/2 w-[200px] h-[300px] rounded-2xl bg-cover bg-center shadow-xl transition-all duration-500"
            style={{ backgroundImage: `url(${src})` }}
          ></div>
        ))}
      </div>

      {/* Buttons */}
      <div className="absolute bottom-5 w-full text-center">
        <button
          onClick={handlePrev}
          className="w-10 h-8 border rounded mr-2 hover:bg-gray-300"
        >
          ←
        </button>
        <button
          onClick={handleNext}
          className="w-10 h-8 border rounded hover:bg-gray-300"
        >
          →
        </button>
      </div>
    </div>
  );
}
