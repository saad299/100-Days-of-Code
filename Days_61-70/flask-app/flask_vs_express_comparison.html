Side-by-side comparison of Express.js and Flask code patterns

Node.js + Express
// install: npm install express

const express = require('express')
const app = express()

app.listen(3000, () => {
  console.log('Running on port 3000')
})
Python + Flask
# install: pip install flask

from flask import Flask

app = Flask(__name__)

if __name__ == '__main__':
    app.run(port=3000)
Node.js + Express
app.get('/', (req, res) => {
  res.send('Hello World')
})

app.post('/users', (req, res) => {
  res.json({ status: 'created' })
})

app.put('/users', (req, res) => { ... })
app.delete('/users', (req, res) => { ... })
Python + Flask
@app.route('/')
def home():
    return 'Hello World'

@app.route('/users', methods=['POST'])
def create_user():
    return { 'status': 'created' }

# methods=['PUT'] / methods=['DELETE']
Node.js + Express
app.get('/users/:id', (req, res) => {
  const { id } = req.params
  res.send(`User ${id}`)
})
Python + Flask
@app.route('/users/<id>')
def get_user(id):
    return f'User {id}'

# typed: /users/<int:id>
Node.js + Express
// GET /search?q=flask&page=2

app.get('/search', (req, res) => {
  const { q, page } = req.query
  res.json({ q, page })
})
Python + Flask
# GET /search?q=flask&page=2

from flask import request

@app.route('/search')
def search():
    q = request.args.get('q')
    page = request.args.get('page')
    return { 'q': q, 'page': page }
Node.js + Express
// parse JSON bodies
app.use(express.json())

app.post('/login', (req, res) => {
  const { email, password } = req.body
  res.json({ email })
})
Python + Flask
# Flask parses JSON automatically
from flask import request

@app.route('/login', methods=['POST'])
def login():
    data = request.get_json()
    email = data['email']
    return { 'email': email }
Node.js + Express
// global middleware
app.use((req, res, next) => {
  console.log(req.method, req.url)
  next()
})

// route-level middleware
app.get('/dashboard', authMiddleware, handler)
Python + Flask
# runs before every request
@app.before_request
def log_request():
    print(request.method, request.path)

# route-level: use a decorator
@app.route('/dashboard')
@login_required
def dashboard(): ...
Node.js + Express
// serve files from /public folder
app.use(express.static('public'))

// access: http://localhost:3000/image.png
Python + Flask
# Flask serves /static folder by default
app = Flask(__name__)

# access: http://localhost:5000/static/image.png
# custom folder:
app = Flask(__name__,
    static_folder='public')
Node.js + Express
// 4-arg signature = error handler
app.use((err, req, res, next) => {
  res.status(500).json({
    error: err.message
  })
})

// 404
app.use((req, res) => {
  res.status(404).send('Not found')
})
Python + Flask
from flask import jsonify

@app.errorhandler(500)
def server_error(err):
    return jsonify({ 'error': str(err) }), 500

# 404
@app.errorhandler(404)
def not_found(err):
    return 'Not found', 404
